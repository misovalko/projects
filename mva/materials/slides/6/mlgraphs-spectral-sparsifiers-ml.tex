Randomized algorithm that finds $\bs$:
\pause 
\begin{itemize}%\addtolength{\itemsep}{0.75\baselineskip}

\item[] \hspace*{-.75cm}Reweigh $s_i = \tfrac{1}{p_i}$  unbiased $\mathbb{E}[\sum_i s_i \bv_i \bv_i] = \sum_e s_e p_e \bv_e \bv_e = \sum_e \bv_e \bv_e$ \pause 
\end{itemize}

\[
1 - \eps \prec \lambda\left(\sum_e s_e \bv_e \bv_e  \transpose \right) \prec 1+\eps

\begin{frame}
\frametitle{Ridge Spectral Graph Sparsification: Intuition}
\begin{definition}\label{def:exact-lev-scores}
\normalfont
\textbf{$\gamma$-effective resistance:}
$\hleb{r_{e}(\gamma)} = \:b_e^\transp(\laplacian_{G} + \gamma\:I)^{-1}\:b_e$
%\vspace{0.2in}

\hbskip
\textbf{Effective dim.:}
$\hlbg{\deff( \gamma )} = \sum_{e} r_{e}(\gamma) 
	= \hlbb{\sum_{i=1}^n \frac{\lambda_i(\laplacian_G)}{\lambda_i(\laplacian_G) + \gamma}}
\hleg{\vphantom{\sum_{i=1}^n(\laplacian_G)}\leq n}$
\end{definition}

\hbskip
\pause
Interpretation as inverse of alternative paths (mostly) lost

\pause
\hbskip
Most existing graph algorithms inapplicable \greycite{kyng2016framework}

Most existing RLA algorithms too slow \greycite{cohen2017input}

\pause
\hbskip
Adapt SOA algorithm for kernel matrix approximation\\
{\small \textcolor{gray}{\textsc{SQUEAK}, \cite{calandriello2017distributed}}}

\begin{frame}
\frametitle{Spectral Graph Sparsification}

We want to make this algorithm fast.
\pause 
\questionbox{How can we compute the effective resistances?}
\moveb
\pause 
Solve a linear system $\wh{\bx} = \argmin_{\bx} \norm{\bL_{G}\bx - \bb_e}$
and then $R_{\text{eff}} = \bb_e \transpose \wh{\bx}$
\pause
\begin{align*}
\action<+->{\text{Gaussian Elimination} \qquad  & \cO(n^3)}\\
\action<+->{\text{Fast Matrix Multiplication}  \qquad  & \cO(n^{2.37})}\\
\action<+->{\text{Spielman \& Teng (2004)} \qquad  & \cO(m \log^{30}n)}\\
\action<+->{\text{Koutis, Miller, and Peng (2010)} \qquad & \cO(m \log n)}
\end{align*}
\vspace{-1\baselineskip}
\begin{itemize}[<+->]
\item Fast solvers for SDD systems:
\consindent{\yellbox{use sparsification internally}\\*
\notecol{\tiny\qquad all the way until you hit the turtles}}
%\item[] \questionbox{still unfeasible when $m$ is large}}
\end{itemize}
\end{frame}
\begin{frame}
\frametitle{Spectral Graph Sparsification}
\questionbox{Chicken and egg problem}
\begin{itemize}
\item[] {\small We need $R_{\text{eff}}$ to compute a sparsifier  $H$ \includegraphics[height=0.7\baselineskip]{../img/arrow_list_rot}}
\consindent{We need a sparsifier $H$ to compute $R_{\text{eff}}$}
\end{itemize}
\pause
\vspace{\baselineskip}
Sampling according to approximate effective resistances $R_{\text{eff}} \leq \wt{R}_{\text{eff}} \leq \alpha R_{\text{eff}}$ give approximate sparsifier $\bL_{G} \preceq \bL_{H} \preceq \alpha\kappa \bL_{G}$
\pause
\vspace{\baselineskip}

Start with poor approximation $\wt{R}_{\text{eff}} \sim 1/n$ and poor sparsifier $n \bI$.
\pause
\begin{itemize}
\item[]  {\small Use $\wt{R}_{\text{eff}}$ to compute an improved approximate sparsifier $H$ \includegraphics[height=0.7\baselineskip]{../img/arrow_list_rot}}
\consindent{Use the sparsifier $H$ to compute improved approximate $\wt{R}_{\text{eff}}$}
\end{itemize}

\pause
\vspace{\baselineskip}
Computing $\wt{R}_{\text{eff}}$ using the sparsifier is fast ($m = \bO(n\log(n))$),
and not too many iterations are necessary.
\end{frame}
