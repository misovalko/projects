\begin{block}{}%<1-| handout:1->%{Submodular function}
  A \textbf{set function} on a discrete set $A$ is \textbf{submodular} if for any $S\subseteq T \subseteq A$  
and for any $e \in A \setminus T$ 
\[
 f(S \cup \{ e \})  - f(S) \geq f(T \cup \{ e \})  - f(T)
\]
\end{block}
\end{frame}

\begin{block}{}%<4| handout:4>{}%{Characterization of the greedy solution}
Let $S^\star = \argmax_{S\subseteq A, |S|\leq k} f(S)$ where $f$
is monotonic and submodular set function
and let $S_{\greedy}$ be a \textbf{greedy solution}. 
\[
 \text{Then \quad} f(S_{\greedy}) \geq \left(1-\tfrac{1}{e}\right)\cdot f(S^\star).
\]
\end{block}

\begin{frame}
\frametitle{Submodularity:  \greedy algorithm}  
 \begin{algorithmic}[1]
{\footnotesize
  \STATE{\bfseries Input:} 
  \STATE \quad $k$: the maximum allowed cardinality of the output
  \STATE \quad $V$: a ground set 
  \STATE \quad $f$: a monotone, non-negative, and submodular function
  \STATE {\bfseries Run:}
  \STATE  $S_0$ = \emptyset
  \FOR{$i = 1$ {\bfseries to} $k$}
\STATE  \alert<0>{$S_i \gets S_{i-1} \cup \left\{ \argmax_{a \in V \setminus S_{i-1} }  \left[ f\left(\left\{ a \right\} \cup S_{i-1} \right)  - f\left( S_{i-1}\right)  \right]\right\}$}
\ENDFOR
 \STATE{\bfseries Output:} 
 \STATE  \quad Return $S_{\greedy} = S_k$
}
\begin{block}{}%<4| handout:4>{}%{Characterization of the greedy solution}
Let $S^\star = \argmax_{S\subseteq A, |S|\leq k} f(S)$ where $f$
is monotonic and submodular set function
and let $S_{\greedy}$ be a \textbf{greedy solution}. 
\[
 \text{Then \quad} f(S_{\greedy}) \geq \left(1-\tfrac{1}{e}\right)\cdot f(S^\star).
\]
\end{block}
\end{algorithmic}
\end{frame}


\begin{frame}
\frametitle{Submodularity:  Approximation guarantee of \greedy}  
Let $S_i$ be the $i$-th set selected by \greedy. We show
\begin{align*}
&f\left(S^\star\right) - f\left(S_{i-1}\right) 
\leq  f\left(S^\star \cup S_{i-1} \right)  - f\left(S_{i-1}\right)  \\ %by monotinicity
\onslide<2->{&\leq  f(a \cup S_{i-1}) - f(S_{i-1}) + f(S^\star / a \cup S_{i-1}) - f(S_{i-1})}\\   %telescoping + submodularit}
\onslide<3->{&\leq  \sum_{a \in S^\star \setminus S_{i-1} } \left( f\left( \{ a \} \cup S_{i-1} \right)  - f\left( S_{i-1}\right)  \right)} \\   %telescoping + submodularit}
\onslide<4->{&\leq  \sum_{a \in S^\star \setminus S_{i-1} } \left(  f\left(  S_{i} \right)  - f\left( S_{i-1}\right) \right)   %definition of the greedy algorithm}
\leq  k \left( f\left(  S_{i} \right)  - f\left( S_{i-1}\right) \right)}
\end{align*}
\onslide<5->{Difference from the optimum of $S_{\greedy} = S_k$ after the $k$-th step \dots
\begin{align*}
 f(S^\star) - f(S_k) 
 	&=  f(S^\star) - f(S_{k-1}) -  \left( f\left(  S_{k} \right)  - f\left( S_{k-1}\right) \right) \\
 	&\leq f(S^\star) - f(S_{k-1})  - \frac{f(S^\star) - f(S_{k-1})}{k}\\
&\leq  \left(1-\tfrac{1}{k}\right)\cdot (f(S^\star) - f(S_{k-1})) \leq \left(1-\tfrac{1}{k}\right)^k\cdot f(S^\star)
\end{align*}}
\end{frame}
