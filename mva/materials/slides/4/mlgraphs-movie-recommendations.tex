\begin{frame}
\frametitle{Use of Laplacians: Movie recommendation}
\questionbox{How to compute the \textbf{${\rm score} (v,m)$}?}
\pause
Using some \gcol{graph distance}!
\pause
\begin{block}{Idea$_1$: maximally weighted path}
${\rm score} (v,m) = \max_{vPm} {\rm weight}(P) = \max_{vPm} \sum_{e\in P} {\rm ranking}(e)$
\end{block}
\pause
\onslide<handout:0>{\notecol{%
Problem: If there is a weak edge, the path should not be good.}}
\pause
\begin{block}{Idea$_2$: change the path weight}
${\rm score}_2 (v,m) = \max_{vPm} {\rm weight}_2(P) = \max_{vPm} \min_{e\in P} {\rm ranking}(e)$
\end{block}
\pause
\onslide<handout:0>{\notecol{%
Problem of 1\&2: Additional paths does not improve the score. }}
\pause
\begin{block}{Idea$_3$: consider everything}
${\rm score}_3 (v,m) = $ max flow from $m$ to $v$
\end{block}
\pause
\onslide<handout:0>{\notecol{%
Problem of 3: Shorter paths do not improve the score. }}
\end{frame}
